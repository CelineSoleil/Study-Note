//Linux系统
#include <iostream>
#include <cstring>
#include <cstdlib>
#include <unistd.h>
#include <arpa/inet.h>
#include <sys/socket.h>

#define PORT 2000
#define BUFFER_SIZE 1024

int main() {
    int server_fd;
    int client_fd;
    struct sockaddr_in server_addr;
    struct sockaddr_in client_addr;
    socklen_t client_addr_len = sizeof(client_addr);
    char buffer[BUFFER_SIZE];

    // 1. 创建服务器 Socket
    server_fd = socket(AF_INET, SOCK_STREAM, 0);
    if (server_fd == -1) {
        perror("socket failed");
        exit(EXIT_FAILURE);
    }

    // 允许重新启动服务器时复用端口
    int reuse = 1;

    if (setsockopt(server_fd,
                   SOL_SOCKET,
                   SO_REUSEADDR,
                   &reuse,
                   sizeof(reuse)) == -1) {
        perror("setsockopt failed");
        close(server_fd);
        exit(EXIT_FAILURE);
    }

    // 2. 设置服务器地址
    memset(&server_addr, 0, sizeof(server_addr));

    server_addr.sin_family = AF_INET;
    server_addr.sin_addr.s_addr = htonl(INADDR_ANY);
    server_addr.sin_port = htons(PORT);

    // 3. 绑定 IP 地址和端口
    if (bind(server_fd,
             (struct sockaddr *)&server_addr,
             sizeof(server_addr)) == -1) {
        perror("bind failed");
        close(server_fd);
        exit(EXIT_FAILURE);
    }

    // 4. 开始监听
    if (listen(server_fd, 5) == -1) {
        perror("listen failed");
        close(server_fd);
        exit(EXIT_FAILURE);
    }

    std::cout << "服务器已启动，监听端口 "
              << PORT
              << "，等待客户端连接..."
              << std::endl;

    // 5. 等待客户端连接
    client_fd = accept(
        server_fd,
        (struct sockaddr *)&client_addr,
        &client_addr_len
    );

    if (client_fd == -1) {
        perror("accept failed");
        close(server_fd);
        exit(EXIT_FAILURE);
    }

    std::cout << "客户端已连接："
              << inet_ntoa(client_addr.sin_addr)
              << ":"
              << ntohs(client_addr.sin_port)
              << std::endl;

    // 6. 循环接收客户端消息，并原样返回
    while (true) {
        memset(buffer, 0, sizeof(buffer));

        ssize_t recv_len = recv(
            client_fd,
            buffer,
            sizeof(buffer) - 1,
            0
        );

        if (recv_len > 0) {
            buffer[recv_len] = '\0';

            std::cout << "收到客户端消息："
                      << buffer
                      << std::endl;

            // 将收到的消息完整发送回客户端
            ssize_t sent_total = 0;

            while (sent_total < recv_len) {
                ssize_t sent_len = send(
                    client_fd,
                    buffer + sent_total,
                    recv_len - sent_total,
                    0
                );

                if (sent_len == -1) {
                    perror("send failed");
                    close(client_fd);
                    close(server_fd);
                    exit(EXIT_FAILURE);
                }

                sent_total += sent_len;
            }
        } else if (recv_len == 0) {
            std::cout << "客户端已断开连接。"
                      << std::endl;
            break;
        } else {
            perror("recv failed");
            break;
        }
    }

    // 7. 关闭 Socket
    close(client_fd);
    close(server_fd);

    std::cout << "服务器已关闭。" << std::endl;

    return 0;
}
