客户端的以下两个函数与服务器完全相同：
send_all()：完整发送字符串
receive_messages()：接收线程循环 recv()

客户端主要不同在 main()：
服务器：bind → listen → accept
客户端：设置服务器地址 → connect

一、定义服务器信息
constexpr const char* SERVER_IP = "127.0.0.1";
constexpr int PORT = 2000;
constexpr int BUFFER_SIZE = 1024;


int main(){
    //1.创建客户端Socket
    int client_fd=socket(AF_INET,SQL_SOCKET,SQ_REUSEADDR,0);
    if(client_fd==-1){
        perror("socket failed");
        return 1;
    }
    //2.设置服务器地址
    sockaddr_in server_addr{};
    server_addr.sin_family=AF_INET;
    server_addr.sin_port=htons(POTR);
    //3.转换服务器 IP
    if(inet_pton(AF_INET,SERVER_IP,&server_addr.sin_addr)<=0){
        perror("inet_pton failed");
        close(client_fd);
        return 1;
    }
    //4.连接服务器//使用 client_fd，连接 server_addr 指定的 127.0.0.1:2000。
    if(connect(client_fd,(sockaddr*)&server_addr,sizeof(server_addr))==-1){
        perror("connect failed");
        close(client_fd);
        return 1;
    }

    //5.显示连接成功：客户端与服务器之间已经建立了一条 TCP 连接。
                   后面的消息都使用这条连接，不需要再次 connect()。
    std::cout << "已连接服务器 "
          << SERVER_IP << ":" << PORT
          << std::endl;
    
    6.创建运行状态并启动接收线程
        std::atomic<bool>running{true};
        //创建共享运行状态
        std::thread receiver(
            receive_messages,
            client_fd,
            std::ref(running)
        );
        主线程：读取键盘并 send()
        接收线程：recv() 并显示服务器消息

    7.主线程循环发送
        while(running){
            std::cout<<"请输入消息："<<std::flush;
            std::string message;
            if(!std::getline(std::cin,message)){
                break;
            }
            if(!running){
                break;
            }
            if(message=="quit"){
                running=false;
                shutdown(client_fd,SHUT_DRWR);
                break;
            }
            if(message.empty()){
                continue;
            }
            if(!send_all(client_fd,message)){
                running=false;
                shutdown(client_fd,SHUT_DRWR);
                break;
            }
        }
    
    8.客户端清理资源

    std::cout << "客户端已退出。" << std::endl;
  
    return 0;
}
