#include <iostream>
#include <string>
#include <cstring>
#include <atomic>
#include <thread>
#include <unistd.h>
#include <arpa/inet.h>
#include <sys/socket.h>

constexpr const char* SERVER_IP = "127.0.0.1";
constexpr int PORT = 2000;
constexpr int BUFFER_SIZE = 1024;

// 将一条消息完整发送出去
bool send_all(int fd, const std::string& message) {
    size_t sent_total = 0;

    while (sent_total < message.size()) {
        ssize_t n = send(
            fd,
            message.c_str() + sent_total,
            message.size() - sent_total,
            MSG_NOSIGNAL
        );

        if (n <= 0) {
            return false;
        }

        sent_total += n;
    }

    return true;
}

// 接收线程：循环接收对方发来的消息
void receive_messages(int fd, std::atomic<bool>& running) {
    char buffer[BUFFER_SIZE];

    while (running) {
        ssize_t n = recv(fd, buffer, sizeof(buffer), 0);

        if (n > 0) {
            std::string message(buffer, n);

            std::cout << "\n对方：" << message
                      << "\n请输入消息：" << std::flush;
        } else if (n == 0) {
            if (running) {
                std::cout
                    << "\n对方已断开连接，按回车退出。\n"
                    << std::flush;
            }

            running = false;
            shutdown(fd, SHUT_RDWR);
            break;
        } else {
            if (running) {
                perror("recv failed");
            }

            running = false;
            shutdown(fd, SHUT_RDWR);
            break;
        }
    }
}

int main() {
    // 1. 创建客户端 Socket
    int client_fd = socket(AF_INET, SOCK_STREAM, 0);

    if (client_fd == -1) {
        perror("socket failed");
        return 1;
    }

    // 2. 设置要连接的服务器地址
    sockaddr_in server_addr{};

    server_addr.sin_family = AF_INET;
    server_addr.sin_port = htons(PORT);

    if (inet_pton(AF_INET,
                  SERVER_IP,
                  &server_addr.sin_addr) <= 0) {
        perror("inet_pton failed");
        close(client_fd);
        return 1;
    }

    // 3. 连接服务器
    if (connect(client_fd,
                (sockaddr*)&server_addr,
                sizeof(server_addr)) == -1) {
        perror("connect failed");
        close(client_fd);
        return 1;
    }

    std::cout << "已连接服务器 "
              << SERVER_IP << ":" << PORT
              << std::endl;

    std::atomic<bool> running{true};

    // 4. 启动接收线程
    std::thread receiver(
        receive_messages,
        client_fd,
        std::ref(running)
    );

    // 5. 主线程循环读取键盘并发送消息
    while (running) {
        std::cout << "请输入消息：" << std::flush;

        std::string message;

        if (!std::getline(std::cin, message)) {
            break;
        }

        // 等待键盘输入期间，对方可能已经断开
        if (!running) {
            break;
        }

        if (message == "quit") {
            running = false;
            shutdown(client_fd, SHUT_RDWR);
            break;
        }

        if (message.empty()) {
            continue;
        }

        if (!send_all(client_fd, message)) {
            std::cout << "发送失败，连接已断开。\n";
            running = false;
            shutdown(client_fd, SHUT_RDWR);
            break;
        }
    }

    // 6. 结束接收线程并释放资源
    running = false;
    shutdown(client_fd, SHUT_RDWR);

    if (receiver.joinable()) {
        receiver.join();
    }

    close(client_fd);

    std::cout << "客户端已退出。" << std::endl;
    return 0;
}
